# Stable Fast 3D (SF3D) — Third 3D Engine Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add `engine="sf3d"` as a third image-to-3D engine in Wyltek Studio, peer to `hy3d` and `trellis`, surfacing the speed/game-ready niche neither covers.

**Architecture:** SF3D plugs into the existing ComfyUI custom-node path. A new `build_sf3d_workflow()` lives in `backends/comfyui.py` next to `build_trellis_workflow()`. The server gets a new `engine` whitelist entry, five `sf3d_*` form fields, an asymmetric branch in `mode_3d` validation (SF3D has no shape/textured split — it's always one-shot textured), and a `("sf3d","textured")` row in `_3D_INNER_TIMEOUTS`. The frontend gets an `#sf3d-options` form group mirroring `#trellis-options` and toggled by the same engine-from-model dispatch. The model-probe path synthesises a virtual `sf3d:` entry the same way it does for `trellis:`.

**Tech Stack:** Python 3 + FastAPI (`server.py`), aiohttp ComfyUI client (`backends/comfyui.py`), plain JS frontend (`static/js/app.js`), pytest with `@pytest.mark.asyncio` for tests, ComfyUI custom node (fork TBD in Task 1).

**Spec:** `docs/superpowers/specs/2026-05-14-sf3d-3d-engine-design.md`

```python
# SF3D image-to-3D workflow. SF3D is single-shot — no shape→continue→textured
# split like TRELLIS — and produces one UV-unwrapped, optionally delit, PBR-
# textured GLB per call. Inference is <1s on CUDA; on ROCm expect a few
# seconds plus ComfyUI overhead (~30s total). The custom-node fork chosen
# in Task 1 of the implementation plan auto-downloads SF3D weights (~3GB)
# on first use.
_VALID_SF3D_REMESH = ("none", "triangle", "quad")


def build_sf3d_workflow(
    image_filename: str,
    *,
    preset: str = "game-ready",
    texture_resolution: int | None = None,
    target_vertex_count: int | None = None,
    remesh: str | None = None,
    delight: bool = True,
    auto_bg_removal: bool = True,
    file_prefix: str = "3D/wyltek-sf3d",
) -> dict:
    """Build a ComfyUI API-format workflow for Stable Fast 3D image-to-3D.

    Args:
        image_filename: name of an image already in ComfyUI/input/.
        preset: "preview" | "game-ready" | "hero". Sets defaults for the
                three knobs below; any explicit knob override wins.
        texture_resolution: 512 | 1024 | 2048. Overrides preset default.
        target_vertex_count: vertex budget after remesh. Overrides preset.
        remesh: "none" | "triangle" | "quad". Overrides preset.
        delight: if True, the texture-bake step inverse-renders lighting
                 out of the albedo (game-engine friendly). Off only when
                 the user wants baked lighting (rare).
        auto_bg_removal: enable rembg on the preprocess node.
        file_prefix: ComfyUI output path prefix; final GLB lands at
                     ComfyUI/output/<file_prefix>_<n>.glb.
    """
    if remesh is not None and remesh not in _VALID_SF3D_REMESH:
        raise ValueError(
            f"remesh must be one of {_VALID_SF3D_REMESH}, got {remesh!r}"
        )

    resolved = _resolve_sf3d_preset(preset)
    if texture_resolution is not None:
        resolved["texture_resolution"] = texture_resolution
    if target_vertex_count is not None:
        resolved["target_vertex_count"] = target_vertex_count
    if remesh is not None:
        resolved["remesh"] = remesh

    # Node class names are stand-ins — replace with the actual class names
    # captured in Phase 0 Task 1 Step 4 if the chosen fork uses different ones.
    wf: dict = {
        "1": {
            "class_type": "SF3D_LOAD_IMAGE",
            "inputs": {"image": image_filename},
        },
        "2": {
            "class_type": "SF3D_PREPROCESS",
            "inputs": {
                "image": ["1", 0],
                "remove_background": bool(auto_bg_removal),
            },
        },
        "3": {
            "class_type": "SF3D_GENERATE",
            "inputs": {
                "image": ["2", 0],
                "texture_resolution": int(resolved["texture_resolution"]),
                "target_vertex_count": int(resolved["target_vertex_count"]),
                "remesh": str(resolved["remesh"]),
                "delight": bool(delight),
            },
        },
        "4": {
            "class_type": "SF3D_EXPORT_GLB",
            "inputs": {
                "mesh": ["3", 0],
                "filename_prefix": file_prefix,
            },
        },
    }
    return wf
```

```python
    async def generate_sf3d(self, params: dict, output_path: str, on_progress) -> dict:
        """Generate a 3D mesh (.glb) via Stable Fast 3D image-to-3D.

        Mirrors the generate_trellis contract — same image-conditioned input,
        same output_path semantics — so the upstream `_run_job` dispatcher
        can treat sf3d / trellis / hy3d symmetrically.

        Required params:
            reference_images: list[str], at least one (first slot used).
        Optional:
            sf3d_preset: "preview" | "game-ready" | "hero" (default "game-ready").
            sf3d_texture_resolution, sf3d_target_vertex_count, sf3d_remesh:
                explicit overrides on preset defaults.
            sf3d_tweak_delight: bool (default True).
            auto_bg_removal: bool (default True).
        """
        import shutil
        from glob import glob

        ref_images = params.get("reference_images") or []
        if not ref_images:
            raise RuntimeError(
                "SF3D generation requires a reference image — image-conditioned only."
            )

        url = self.url
        client_id = str(uuid.uuid4())

        # SF3D is single-view; ignore slots 1-3 if supplied.
        src = Path(ref_images[0])
        if not src.exists():
            raise RuntimeError(f"Reference image not found: {src}")
        comfy_input = Path("/home/phill/ComfyUI/input")
        ref_name = f"sf3d_{client_id[:8]}{src.suffix or '.png'}"
        shutil.copy2(src, comfy_input / ref_name)

        file_prefix = f"3D/wyltek-sf3d_{client_id[:8]}"
        workflow = build_sf3d_workflow(
            image_filename=ref_name,
            preset=params.get("sf3d_preset", "game-ready"),
            texture_resolution=params.get("sf3d_texture_resolution"),
            target_vertex_count=params.get("sf3d_target_vertex_count"),
            remesh=params.get("sf3d_remesh"),
            delight=bool(params.get("sf3d_tweak_delight", True)),
            auto_bg_removal=bool(params.get("auto_bg_removal", True)),
            file_prefix=file_prefix,
        )

        await on_progress(10, "Submitting SF3D workflow...")

        # Submit + wait via the same websocket helper used by generate_trellis.
        # First run incurs a ~3GB weight download — surface that to the user
        # via the existing progress smoother (handled by _run_job).
        inner_timeout = int(params.get("_inner_timeout_s", 60))
        await self._submit_and_wait(
            workflow=workflow,
            client_id=client_id,
            timeout_s=inner_timeout,
            on_progress=on_progress,
        )

        await on_progress(90, "Locating .glb output...")

        comfy_output_dir = Path("/home/phill/ComfyUI/output")
        pattern = str(comfy_output_dir / f"{file_prefix}*.glb")
        matches = sorted(glob(pattern))
        chosen = matches[-1] if matches else None
        if not chosen:
            raise RuntimeError(
                f"SF3D finished but no .glb produced at {pattern}. "
                "Check ComfyUI logs for an error."
            )
        shutil.copy2(chosen, output_path)

        await on_progress(100, "Done")
        return {
            "filename": Path(output_path).name,
            "format": "glb",
            "mode": "textured",  # SF3D is always textured; mode_3d unused.
        }
```
